`timescale 1ns/1ps

module tb_top;

    logic       clk;
    logic       rst_n;
    logic       we;
    logic       re;
    logic [2:0] addr;
    logic [7:0] wdata;
    logic [7:0] rdata;

    logic [7:0] reference_memory [0:7];

    integer error_count;
    integer test_count;
    integer i;

    logic       random_operation;
    logic [2:0] random_address;
    logic [7:0] random_data;


    ram_8x8 dut (
        .clk   (clk),
        .rst_n (rst_n),
        .we    (we),
        .re    (re),
        .addr  (addr),
        .wdata (wdata),
        .rdata (rdata)
    );


    initial
        clk = 1'b0;

    always #5
        clk = ~clk;


    initial begin
        $dumpfile("dump.vcd");
        $dumpvars(0, tb_top);
    end


    covergroup ram_coverage @(posedge clk);

        option.per_instance = 1;

        operation_cp:
            coverpoint {we, re} iff (rst_n) {
                bins idle_operation  = {2'b00};
                bins read_operation  = {2'b01};
                bins write_operation = {2'b10};
                illegal_bins simultaneous_read_write = {2'b11};
            }

        address_cp:
            coverpoint addr iff (
                rst_n && (we || re)
            ) {
                bins each_address[] = {[0:7]};
            }

        write_data_cp:
            coverpoint wdata iff (
                rst_n && we
            ) {
                bins zero_data = {8'h00};
                bins low_data  = {[8'h01:8'h55]};
                bins mid_data  = {[8'h56:8'hAA]};
                bins high_data = {[8'hAB:8'hFE]};
                bins max_data  = {8'hFF};
            }

        operation_address_cross:
            cross operation_cp, address_cp;

    endgroup


    ram_coverage coverage = new();


    no_simultaneous_read_write:
        assert property (
            @(posedge clk)
            disable iff (!rst_n)
            !(we && re)
        )
        else begin
            error_count++;
            $error(
                "ASSERTION FAILED: we and re active together"
            );
        end


    known_address:
        assert property (
            @(posedge clk)
            disable iff (!rst_n)
            (we || re) |-> !$isunknown(addr)
        )
        else begin
            error_count++;
            $error(
                "ASSERTION FAILED: Address contains X/Z"
            );
        end


    known_write_data:
        assert property (
            @(posedge clk)
            disable iff (!rst_n)
            we |-> !$isunknown(wdata)
        )
        else begin
            error_count++;
            $error(
                "ASSERTION FAILED: Write data contains X/Z"
            );
        end


    reset_clears_rdata:
        assert property (
            @(posedge clk)
            !rst_n |=> (rdata == 8'h00)
        )
        else begin
            error_count++;
            $error(
                "ASSERTION FAILED: Reset did not clear rdata"
            );
        end


    task automatic write_ram(
        input logic [2:0] write_address,
        input logic [7:0] write_data
    );
        begin
            @(negedge clk);

            we    = 1'b1;
            re    = 1'b0;
            addr  = write_address;
            wdata = write_data;

            @(posedge clk);
            #1;

            reference_memory[write_address] =
                write_data;

            test_count++;

            $display(
                "[%0t] WRITE: addr=%0d data=0x%02h",
                $time,
                write_address,
                write_data
            );

            @(negedge clk);

            we = 1'b0;
        end
    endtask


    task automatic read_ram(
        input logic [2:0] read_address
    );
        begin
            @(negedge clk);

            we   = 1'b0;
            re   = 1'b1;
            addr = read_address;

            @(posedge clk);

            #1;

            test_count++;

            if (
                rdata !==
                reference_memory[read_address]
            ) begin

                error_count++;

                $error(
                    "READ FAIL: addr=%0d expected=0x%02h actual=0x%02h",
                    read_address,
                    reference_memory[read_address],
                    rdata
                );

            end
            else begin

                $display(
                    "[%0t] READ PASS: addr=%0d data=0x%02h",
                    $time,
                    read_address,
                    rdata
                );

            end

            @(negedge clk);

            re = 1'b0;
        end
    endtask


    initial begin

        rst_n = 1'b0;
        we    = 1'b0;
        re    = 1'b0;
        addr  = 3'b000;
        wdata = 8'h00;

        error_count = 0;
        test_count  = 0;

        for (i = 0; i < 8; i++)
            reference_memory[i] = 8'h00;


        $display("\n===== RESET TEST =====");

        repeat (3)
            @(posedge clk);

        @(negedge clk);
        rst_n = 1'b1;

        $display("[%0t] Reset released", $time);


        $display("\n===== DIRECTED WRITE TEST =====");

        for (i = 0; i < 8; i++) begin
            write_ram(
                i[2:0],
                8'h11 * (i + 1)
            );
        end


        $display("\n===== DIRECTED READ TEST =====");

        for (i = 0; i < 8; i++)
            read_ram(i[2:0]);


        $display("\n===== BOUNDARY VALUE TEST =====");

        write_ram(3'd0, 8'h00);
        read_ram (3'd0);

        write_ram(3'd7, 8'hFF);
        read_ram (3'd7);


        $display("\n===== RANDOM TEST =====");

        repeat (50) begin

            random_operation =
                $urandom_range(0, 1);

            random_address =
                $urandom_range(0, 7);

            random_data =
                $urandom_range(0, 255);

            if (random_operation)
                write_ram(
                    random_address,
                    random_data
                );
            else
                read_ram(
                    random_address
                );

        end


        #10;

        $display(
            "\n========================================"
        );

        $display("RAM 8x8 VERIFICATION REPORT");

        $display(
            "Total checked tests : %0d",
            test_count
        );

        $display(
            "Total errors        : %0d",
            error_count
        );

        $display(
            "Functional coverage : %0.2f%%",
            coverage.get_inst_coverage()
        );

        if (error_count == 0)
            $display("FINAL RESULT        : TEST PASSED");
        else
            $display("FINAL RESULT        : TEST FAILED");

        $display(
            "========================================\n"
        );

        $finish;
    end


    initial begin
        #100000;

        $fatal(
            1,
            "TEST TIMEOUT"
        );
    end

endmodule
